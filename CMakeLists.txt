cmake_minimum_required(VERSION 3.20)
project(SCHRODR VERSION 0.2.0 LANGUAGES CXX)

# Keep windows.h from defining the min/max macros no matter which header pulls it in.
add_compile_definitions($<$<CXX_COMPILER_ID:MSVC>:NOMINMAX>)
# The tests and tools hold whole keyboards and engines (0.4 to 0.5 MB each) on the stack, as
# Linux's and macOS's 8 MB stacks allow; Windows gives an executable 1 MB, so its programs get
# 16 MB. The plug-in keeps its engine on the heap and is unaffected.
if(MSVC)
  string(APPEND CMAKE_EXE_LINKER_FLAGS " /STACK:16777216")
endif()

option(SCHRODR_COVERAGE "Instrument the tests for line coverage (GCC/Clang)" OFF)
option(SCHRODR_EDITOR "Build the native editor" ON)

function(schrodr_warnings target)
  target_compile_options(${target} PRIVATE
    $<$<CXX_COMPILER_ID:MSVC>:/W4>
    $<$<NOT:$<CXX_COMPILER_ID:MSVC>>:-Wall;-Wextra;-Wpedantic;-Wconversion;-Wno-sign-conversion>)
endfunction()
# The tests use CHECK-style soft assertions plus a few hard asserts; never strip them.
function(schrodr_enable_assertions target)
  target_compile_options(${target} PRIVATE
    $<$<CXX_COMPILER_ID:MSVC>:/UNDEBUG>
    $<$<NOT:$<CXX_COMPILER_ID:MSVC>>:-UNDEBUG>)
endfunction()
function(schrodr_coverage target)
  if(SCHRODR_COVERAGE AND NOT MSVC)
    target_compile_options(${target} PRIVATE --coverage -O0 -g)
    target_link_options(${target} PUBLIC --coverage)
  endif()
endfunction()

# --- third party --------------------------------------------------------------------------------
# The CLAP headers are pinned. Offline builds point FETCHCONTENT_SOURCE_DIR_CLAP at an existing
# checkout (see README).
include(FetchContent)
FetchContent_Declare(clap
  GIT_REPOSITORY https://github.com/free-audio/clap.git
  GIT_TAG 1.2.7
  GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(clap)

# --- the engine ---------------------------------------------------------------------------------
# Everything that makes sound, apart from the CLAP glue and the editor, so the tests can link
# it without loading the plug-in. The DSP is header-only; core.cpp compiles it once as a check.
add_library(schrodr_core STATIC src/core.cpp)
target_compile_features(schrodr_core PUBLIC cxx_std_20)
target_include_directories(schrodr_core PUBLIC src)
set_target_properties(schrodr_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
schrodr_warnings(schrodr_core)
schrodr_coverage(schrodr_core)

# --- the plug-in --------------------------------------------------------------------------------
if(SCHRODR_EDITOR AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
  find_package(X11 REQUIRED)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(XFT REQUIRED IMPORTED_TARGET xft)
endif()

add_library(schrodr_clap MODULE src/schrodr.cpp)
target_compile_features(schrodr_clap PRIVATE cxx_std_20)
target_include_directories(schrodr_clap PRIVATE "${clap_SOURCE_DIR}/include")
target_compile_definitions(schrodr_clap PRIVATE SCHRODR_VERSION="${PROJECT_VERSION}")
target_link_libraries(schrodr_clap PRIVATE schrodr_core)
schrodr_warnings(schrodr_clap)
schrodr_coverage(schrodr_clap)
if(SCHRODR_EDITOR)
  target_compile_definitions(schrodr_clap PRIVATE SCHRODR_EDITOR=1)
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_link_libraries(schrodr_clap PRIVATE X11::X11 X11::Xrender PkgConfig::XFT)
  elseif(WIN32)
    target_link_libraries(schrodr_clap PRIVATE user32 gdi32)
  elseif(APPLE)
    enable_language(OBJCXX)
    set_source_files_properties(src/schrodr.cpp PROPERTIES LANGUAGE OBJCXX)
    target_link_libraries(schrodr_clap PRIVATE "-framework Cocoa")
  endif()
endif()
set_target_properties(schrodr_clap PROPERTIES
  PREFIX "" OUTPUT_NAME "SCHRODR" SUFFIX ".clap"
  CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
if(APPLE)
  set_target_properties(schrodr_clap PROPERTIES BUNDLE TRUE BUNDLE_EXTENSION clap SUFFIX "")
endif()
install(TARGETS schrodr_clap BUNDLE DESTINATION clap LIBRARY DESTINATION clap RUNTIME DESTINATION clap)

# --- tests --------------------------------------------------------------------------------------
enable_testing()

function(schrodr_test name)
  add_executable(${name} tests/${name}.cpp ${ARGN})
  target_compile_features(${name} PRIVATE cxx_std_20)
  target_include_directories(${name} PRIVATE "${clap_SOURCE_DIR}/include" src tests)
  target_link_libraries(${name} PRIVATE schrodr_core)
  schrodr_warnings(${name})
  schrodr_enable_assertions(${name})
  schrodr_coverage(${name})
  add_test(NAME ${name} COMMAND ${name})
  set_tests_properties(${name} PROPERTIES TIMEOUT 900)
endfunction()

schrodr_test(string_tests)
schrodr_test(keyboard_tests)
schrodr_test(bars_tests)
schrodr_test(pipes_tests)
schrodr_test(air_tests)
schrodr_test(engine_tests)

# Suites that load the built .clap and drive it through the public CLAP interfaces, as a host does.
function(schrodr_plugin_test name)
  add_executable(${name} tests/${name}.cpp)
  target_compile_features(${name} PRIVATE cxx_std_20)
  target_include_directories(${name} PRIVATE "${clap_SOURCE_DIR}/include" src tests)
  target_link_libraries(${name} PRIVATE ${CMAKE_DL_LIBS})
  schrodr_warnings(${name})
  schrodr_enable_assertions(${name})
  add_test(NAME ${name} COMMAND ${name} $<TARGET_FILE:schrodr_clap>)
  set_tests_properties(${name} PROPERTIES TIMEOUT 600)
endfunction()
schrodr_plugin_test(plugin_smoke)
schrodr_plugin_test(note_tests)
schrodr_plugin_test(state_tests)
